A bucketing problem: every word belongs to exactly one group, so what do all the words in one group share that no outsider has? If you could write that shared thing down as a label, where would you file each word?

== pseudocode.gleam
pub fn group_anagrams(strs: List(String)) -> List(List(String)) {
  strs
  |> list.fold(dict.new(), fn(acc, s) {
    dict.upsert(acc, signature(s), fn(group) {
      case group {
        option.Some(members) -> [s, ..members]
        option.None -> [s]
      }
    })
  })
  |> dict.values
  |> list.map(list.reverse)
}

fn signature(word: String) -> String {
  string.to_graphemes(word)
  |> list.fold(dict.new(), fn(acc, g) {
    dict.upsert(acc, g, fn(n) { option.unwrap(n, 0) + 1 })
  })
  |> dict.to_list
  |> list.sort(fn(a, b) { string.compare(a.0, b.0) })
  |> list.map(fn(entry) { entry.0 <> int.to_string(entry.1) })
  |> string.concat
}

const groups = new Map<string, string[]>();
for (const s of strs) {
  const tally = new Array(26).fill(0);
  for (const char of s) tally[char.charCodeAt(0) - 97]++;
  const key = tally.join(",");
  const group = groups.get(key);
  if (group) {
    group.push(s);
  } else {
    groups.set(key, [s]);
  }
}

groups := map[[26]int][]string{}
order := [][26]int{}
for _, s := range strs {
    var key [26]int
    for i := 0; i < len(s); i++ {
        key[s[i]-'a']++
    }
