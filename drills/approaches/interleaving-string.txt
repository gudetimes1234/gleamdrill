== pseudocode
if len(s1) + len(s2) != len(s3): return false
memo = empty map
works(i, j):
    if i == len(s1) and j == len(s2): return true
    if (i, j) not in memo:
        target = s3[i + j]
        ok = false
        if i < len(s1) and s1[i] == target and works(i + 1, j): ok = true
        if j < len(s2) and s2[j] == target and works(i, j + 1): ok = true
        memo[(i, j)] = ok
    return memo[(i, j)]
return works(0, 0)
