memo = empty map
works(i, j):
    if j >= len(p): return i >= len(s)
    if (i, j) not in memo:
        here = i < len(s) and (p[j] == s[i] or p[j] == ".")
        if j + 1 < len(p) and p[j+1] == "*":
            memo[(i, j)] = works(i, j + 2) or (here and works(i + 1, j))
        else:
            memo[(i, j)] = here and works(i + 1, j + 1)
    return memo[(i, j)]
