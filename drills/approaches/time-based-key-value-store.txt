== pseudocode
set(key, value, timestamp):
    store[key].append((timestamp, value))

get(key, timestamp):
    history = store[key], or empty list
    low = 0
    high = len(history) - 1
    best = ""
    while low <= high:
        mid = (low + high) / 2
        if history[mid].time <= timestamp:
            best = history[mid].value
            low = mid + 1
        else:
            high = mid - 1
    return best
