== pseudocode
seen = empty set
for r in 0..8:
    for c in 0..8:
        if board[r][c] == ".": continue
        v = board[r][c]
        for key in [(v, "row", r), (v, "col", c), (v, "box", r div 3, c div 3)]:
            if key in seen: return false
            seen.add(key)
return true
